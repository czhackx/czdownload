(() => {
    // Cleanup ของเก่า
    document.querySelectorAll("#nc-btn-container").forEach(e => e.remove());
    if (window.__ncGarageInterval) clearInterval(window.__ncGarageInterval);
    if (window.__ncGarageObserver) window.__ncGarageObserver.disconnect();

    let deleteMode = false;
    let macroMode = false;
    let vehiclesCache = [];
    const selectedPlates = new Set();
    const deletedPlates = new Set();

    function isGarageOpen() {
        // อัปเดตตามโครงสร้างใหม่
        const garage = document.querySelector(".nc-garage.show");
        return !!garage;
    }

    function closeUI() {
        document.dispatchEvent(new KeyboardEvent("keydown", { key: "Escape", code: "Escape", keyCode: 27, bubbles: true }));
    }

    async function postOpenStorage(plate) {
        try {
            await fetch("https://nc_garage/OpenStorage", {
                method: "POST",
                headers: { "Content-Type": "application/json; charset=UTF-8" },
                body: JSON.stringify({ type: "trunk", vehicleType: "car", plate: plate })
            });
            setTimeout(closeUI, 150);
        } catch (e) { console.log(e); }
    }

    function scanVehicles() {
        vehiclesCache = [];
        // ปรับ Selector ให้ตรงกับ UI ใหม่
        document.querySelectorAll(".nc-vehicle").forEach(vehicle => {
            const plate = vehicle.querySelector(".nc-vehicle__data__plate")?.innerText?.trim();
            const model = vehicle.querySelector(".nc-vehicle__data__title")?.innerText?.trim();
            if (!plate || !model || deletedPlates.has(plate)) return;
            vehiclesCache.push({ plate, model });
        });
    }

    function createButton(text, color, onclick) {
        const btn = document.createElement("button");
        btn.innerText = text;
        Object.assign(btn.style, { background: color, color: "#fff", border: "none", borderRadius: "6px", padding: "6px 12px", cursor: "pointer", fontSize: "12px" });
        btn.onclick = onclick;
        return btn;
    }

    function renderButtons() {
        // อัปเดตตำแหน่งการวางปุ่ม (Anchor Point)
        const storage = document.querySelector(".nc-detail__detail-top__data__storage");
        if (!storage) return;

        const old = document.querySelector("#nc-btn-container");
        if (old) old.remove();

        const container = document.createElement("div");
        container.id = "nc-btn-container";
        Object.assign(container.style, { marginTop: "12px", padding: "10px", borderRadius: "10px", background: "rgba(0,0,0,.35)" });

        // ปุ่มควบคุม
        const top = document.createElement("div");
        Object.assign(top.style, { display: "flex", gap: "8px", marginBottom: "10px", flexWrap: "wrap" });
        top.appendChild(createButton(deleteMode ? "Exit" : "Del", "#ef4444", () => { deleteMode = !deleteMode; renderButtons(); }));
        top.appendChild(createButton("Reset", "#3b82f6", () => { deletedPlates.clear(); selectedPlates.clear(); scanVehicles(); renderButtons(); }));
        top.appendChild(createButton("Macro", macroMode ? "#22c55e" : "#10b981", () => { macroMode = !macroMode; renderButtons(); }));

        const list = document.createElement("div");
        Object.assign(list.style, { display: "flex", flexWrap: "wrap", gap: "6px", maxHeight: "220px", overflowY: "auto" });

        vehiclesCache.forEach(vehicle => {
            const btn = document.createElement("button");
            const selected = selectedPlates.has(vehicle.plate);
            btn.innerText = deleteMode ? vehicle.model : (macroMode ? " " : vehicle.plate);
            Object.assign(btn.style, {
                width: deleteMode ? "100px" : "50px", height: "34px",
                background: deleteMode ? "#facc15" : (selected ? "#4ade80" : "#e5e7eb"),
                color: "#000", border: "none", borderRadius: "6px", cursor: "pointer", fontSize: "11px"
            });
            btn.onclick = () => {
                if (deleteMode) {
                    deletedPlates.add(vehicle.plate);
                    vehiclesCache = vehiclesCache.filter(x => x.plate !== vehicle.plate);
                    renderButtons();
                    return;
                }
                selectedPlates.has(vehicle.plate) ? selectedPlates.delete(vehicle.plate) : selectedPlates.add(vehicle.plate);
                renderButtons();
                postOpenStorage(vehicle.plate);
            };
            list.appendChild(btn);
        });

        container.appendChild(top);
        container.appendChild(list);
        storage.appendChild(container);
    }

    function mainLoop() {
        if (!isGarageOpen()) return;
        if (!document.querySelector("#nc-btn-container")) {
            scanVehicles();
            renderButtons();
        }
    }

    window.__ncGarageInterval = setInterval(mainLoop, 500);
})();
